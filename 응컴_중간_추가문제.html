1/* 유형 B 실제 문제 (족보, 2025 중간 2번) */2#include <stdio.h>3 4int main(void) {5    int n, p, cnt, first;6 7    while (1) {8        printf("Enter an integer>> ");9        if (scanf("%d", &n) != 1 || n == 0) break;10 11        first = 1;12        printf("  ");13        /* p * p <= n 인 동안만 나누어 보면 된다 */14        for (p = 2; p * p <= n; p++) {15            cnt = 0;16            while (n % p == 0) {17                n /= p;18                cnt++;19            }20            if (cnt > 0) {21                if (!first) printf("*");22                printf("%d^%d", p, cnt);23                first = 0;24            }25        }26        /* 남은 수가 1보다 크면 그 수도 소수다 */27        if (n > 1) {28            if (!first) printf("*");29            printf("%d^1", n);30        }31        printf("\n");32    }33    return 0;34}

1/* 유형 D 실제 문제 (족보) */2#include <stdio.h>3#include <string.h>4 5int main(void) {6    char s[1000];7    int depth, i, ok;8 9    while (1) {10        printf("Enter a parens>> ");11        if (scanf("%999s", s) != 1) break;12        if (strcmp(s, "quit") == 0) break;13 14        /* depth: 닫히지 않은 '('의 개수 */15        depth = 0;16        ok = 1;17        for (i = 0; s[i] != '\0'; i++) {18            if (s[i] == '(') {19                depth++;20            } else if (s[i] == ')') {21                depth--;22                if (depth < 0) {23                    ok = 0;24                    break;25                }26            }27        }28        /* 열린 괄호가 남아 있어도 짝이 맞지 않는다 */29        if (depth != 0) ok = 0;30        printf("Result >> %s\n", ok ? "Matched" : "Unmatched");31    }32    return 0;33}

  

  

    
필요한 헤더
헤더하는 일이 코드에서 쓰는 함수
<stdio.h>입출력printf, scanf
<string.h>문자열 다루기strcmp

    
조건이 바뀌면
1
괄호가 세 종류
개수로는 ([)]를 못 가린다. 여는 괄호를 문자 스택에 넣는다
2
개수가 같아도 틀린다
)(는 첫 문자에서 depth가 -1이 된다

응용컴퓨터프로그래밍 중간고사 추가 문제
유형 D · 응용 문제
세 종류의 괄호 검사


문자 ‘(’, ‘)’, ‘[’, ‘]’, ‘{’, ‘}’만을 이용한 문자열을 입력하여, 괄호의 짝이 맞는지 확인하는 프로그램을 작성하라. 여는 괄호와 닫는 괄호는 종류가 같아야 짝이 된다. “quit”를 입력하면 종료하도록 한다.


실행 예


Enter a parens>> {[()()]}
Result >> Matched
Enter a parens>> ([)]
Result >> Unmatched
Enter a parens>> [{}]()
Result >> Matched
Enter a parens>> quit



  
  
  • 괄호가 세 종류다
  • 개수만 세면 ([)]를 가려낼 수 없다. 여는 괄호를 문자 스택에 넣고, 닫는 괄호가 나오면 맨 위가 같은 종류인지 확인한다
